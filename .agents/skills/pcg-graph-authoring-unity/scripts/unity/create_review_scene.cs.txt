// create_review_scene.cs.txt
// Paste into user-unity execute_csharp_script after replacing tokens.
// Tokens: __REVIEW_SLUG__
//
// Opens a brand-new Empty scene and saves it under the fixed preview dir:
//   Assets/PICG-Workspace/Scenes/PcgReview_<slug>.scene
// Never ask the user for a scene path — this directory is the SSOT.
// Prefer this over manage_scene create when path nesting is unreliable.

using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine.SceneManagement;

string slug = "__REVIEW_SLUG__";
string dir = "Assets/PICG-Workspace/Scenes";
string path = dir + "/PcgReview_" + slug + ".scene";

if (!AssetDatabase.IsValidFolder("Assets/PICG-Workspace"))
{
    if (!AssetDatabase.IsValidFolder("Assets"))
        return "FAIL Assets missing";
    AssetDatabase.CreateFolder("Assets", "PICG-Workspace");
}
if (!AssetDatabase.IsValidFolder(dir))
    AssetDatabase.CreateFolder("Assets/PICG-Workspace", "Scenes");

var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
if (!EditorSceneManager.SaveScene(scene, path))
    return "FAIL save " + path;

return "OK " + path + " roots=" + scene.rootCount;
